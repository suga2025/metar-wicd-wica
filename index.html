// WORKER FINAL - BMKG + AviationWeather + Proxy Gambar const BMKG_BASE='https://web-aviation.bmkg.go.id/web/'; const PAGES={metar:BMKG_BASE+'metar_speci.php',taf:BMKG_BASE+'taf.php'}; const UA={'User-Agent':'Mozilla/5.0 (Windows NT 10.0; Win64; x64) Chrome/130.0.0.0 Safari/537.36'}; const CORS={'Access-Control-Allow-Origin':'*','Access-Control-Allow-Methods':'GET,POST,OPTIONS','Access-Control-Allow-Headers':'*','Access-Control-Max-Age':'86400'}; function iso(d){return d.toISOString().slice(0,16);} async function getSession(kind){ const r=await fetch(PAGES[kind],{headers:UA}); const html=await r.text(); const tokenMatch=html.match(/name=["']_token["'][^>]*value=["']([^"']+)["']/i); const setCookie=r.headers.get('set-cookie')||''; return {token: tokenMatch?tokenMatch[1]:'', cookie: setCookie.split(';')[0]}; } async function fetchBMKG(kind,icao,hours){ const sess=await getSession(kind); const now=new Date(); const from=new Date(now.getTime()-3600000*hours); const to=new Date(now.getTime()+43200000); const params=new URLSearchParams({_token:sess.token||'', stasiun:icao, from:iso(from), to:iso(to)}); if(kind==='metar'){params.append('metar','SA'); params.append('speci','SP');} const res=await fetch(PAGES[kind],{method:'POST',headers:{...UA,'Content-Type':'application/x-www-form-urlencoded','Referer':PAGES[kind],'Origin':'https://web-aviation.bmkg.go.id','X-Requested-With':'XMLHttpRequest',Cookie:sess.cookie},body:params}); return await res.text(); } export default{ async fetch(req){ const url=new URL(req.url); if(req.method==='OPTIONS') return new Response('',{headers:CORS}); // 1. Proxy gambar / umum const target=url.searchParams.get('url'); if(target){const r=await fetch(target,{headers:UA}); return new Response(await r.arrayBuffer(),{headers:{...CORS,'Content-Type':r.headers.get('content-type')||'text/plain'}});} // 2. Proxy aviationweather lama (fallback) const ids=url.searchParams.get('ids'); if(ids){const r=await fetch(`https://aviationweather.gov/api/data/metar?ids=${ids}&format=json`,{headers:UA}); return new Response(await r.text(),{headers:{...CORS,'Content-Type':'application/json'}});} // 3. Proxy BMKG const icao=url.searchParams.get('icao')||'WICD', kind=url.searchParams.get('kind')||'metar', hours=url.searchParams.get('hours')||'24'; try{const html=await fetchBMKG(kind,icao,parseInt(hours)); return new Response(html,{headers:{...CORS,'Content-Type':'text/html'}});}catch(e){return new Response('Error: '+e.message,{status:500,headers:CORS});} } }
